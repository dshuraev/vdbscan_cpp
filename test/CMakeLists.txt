include(FetchContent)

FetchContent_Declare(
    Catch2
    GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG v3.8.0
    GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(Catch2)

include(CTest)
list(APPEND CMAKE_MODULE_PATH ${Catch2_SOURCE_DIR}/extras)
include(Catch)

# ── Unit tests ────────────────────────────────────────────────────────────────

add_executable(vdbscan_tests test_vdbscan.cpp)
target_link_libraries(vdbscan_tests PRIVATE vdbscan Catch2::Catch2WithMain)
catch_discover_tests(vdbscan_tests)

# ── Synthetic integration tests ───────────────────────────────────────────────
# Requires PLY files generated by 'task gen:synthetic'.
# Enable with: -DVDBSCAN_TEST_SYNTHETIC=ON -DSYNTHETIC_DIR=<path>

option(VDBSCAN_TEST_SYNTHETIC
    "Build integration tests against synthetic PLY ground-truth data" OFF)

if(VDBSCAN_TEST_SYNTHETIC)
    if(NOT DEFINED SYNTHETIC_DIR OR SYNTHETIC_DIR STREQUAL "")
        message(FATAL_ERROR
            "VDBSCAN_TEST_SYNTHETIC requires -DSYNTHETIC_DIR=<path to .ply files>")
    endif()

    cmake_path(ABSOLUTE_PATH SYNTHETIC_DIR
               BASE_DIRECTORY "${CMAKE_SOURCE_DIR}"
               OUTPUT_VARIABLE SYNTHETIC_DIR_ABS)

    add_executable(vdbscan_synth_tests test_synthetic.cpp)
    target_link_libraries(vdbscan_synth_tests PRIVATE vdbscan Catch2::Catch2WithMain)
    target_compile_definitions(vdbscan_synth_tests PRIVATE
        SYNTHETIC_DIR="${SYNTHETIC_DIR_ABS}")

    catch_discover_tests(vdbscan_synth_tests)
endif()

# ── Coverage instrumentation ──────────────────────────────────────────────────

option(VDBSCAN_COVERAGE
    "Instrument binaries for gcov/llvm-cov line and branch coverage" OFF)

if(VDBSCAN_COVERAGE)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        set(_cov_targets vdbscan vdbscan_tests)
        if(VDBSCAN_TEST_SYNTHETIC)
            list(APPEND _cov_targets vdbscan_synth_tests)
        endif()
        foreach(tgt IN LISTS _cov_targets)
            target_compile_options(${tgt} PRIVATE --coverage)
            target_link_options(${tgt} PRIVATE --coverage)
        endforeach()
    else()
        message(WARNING "VDBSCAN_COVERAGE is only supported with Clang or GCC")
    endif()
endif()
